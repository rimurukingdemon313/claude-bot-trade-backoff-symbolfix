# The trading bot runs on the Python standard library alone. Everything in
# bot/ — the broker client, SMC engine, risk engine, executor, scheduler and
# HTTP service — uses only json, urllib, sqlite3, threading, dataclasses and
# friends. That is deliberate: fewer dependencies means less to break on a
# platform redeploy and a smaller supply-chain surface for a process that
# holds broker credentials.
#
# The one optional dependency:
#
#   psycopg  — required ONLY when DATABASE_URL is set (PostgreSQL
#              persistence). Railway's filesystem is ephemeral, so the kill
#              switch, daily-loss counters and execution intents must live
#              in Postgres for a production deployment. If DATABASE_URL is
#              set and this driver is missing, the bot refuses to start
#              rather than silently downgrading to storage that cannot
#              survive the next deploy.
psycopg[binary]>=3.1,<4

# Test-only. Not imported by anything in bot/.
pytest>=8.0
